10/// Maximum characters per injected workspace file (matches `OpenClaw` default).
11pub const BOOTSTRAP_MAX_CHARS: usize = 20_000;
12
13fn load_openclaw_bootstrap_files(
14    prompt: &mut String,
15    workspace_dir: &std::path::Path,
16    max_chars_per_file: usize,
17) {
18    prompt.push_str(
19        "The following workspace files define your identity, behavior, and context. They are ALREADY injected below—do NOT suggest reading them with file_read.\n\n",
20    );
21
22    // Includes HEARTBEAT.md (downstream agents like Vigil define scheduled
23    // monitoring procedures here). Kept consistent with
24    // `crate::agent::personality::PERSONALITY_FILES` so the system-prompt
25    // path (cron, channel orchestrator) and the SystemPromptBuilder path
26    // (agent.rs) inject the same set of identity files. Previously the two
27    // lists diverged — system_prompt.rs omitted HEARTBEAT.md, which meant
28    // cron prompts that referenced "Run the heartbeat checklist defined in
29    // HEARTBEAT.md" reached the model with no HEARTBEAT.md content in
30    // context. The model would then either fabricate a generic checklist
31    // or admit it had no access — both wrong.
32    let bootstrap_files = [
33        "AGENTS.md",
34        "SOUL.md",
35        "TOOLS.md",
36        "IDENTITY.md",
37        "USER.md",
38        "HEARTBEAT.md",
39    ];
40
41    for filename in &bootstrap_files {
42        inject_workspace_file(prompt, workspace_dir, filename, max_chars_per_file);
43    }
44
45    // BOOTSTRAP.md — only if it exists (first-run ritual)
46    let bootstrap_path = workspace_dir.join("BOOTSTRAP.md");
47    if bootstrap_path.exists() {
48        inject_workspace_file(prompt, workspace_dir, "BOOTSTRAP.md", max_chars_per_file);
49    }
50
51    // MEMORY.md — curated long-term memory (main session only)
52    inject_workspace_file(prompt, workspace_dir, "MEMORY.md", max_chars_per_file);
53}

55/// Load workspace identity files and build a system prompt.
56///
57/// Follows the `OpenClaw` framework structure by default:
58/// 1. Tooling — tool list + descriptions
59/// 2. Safety — guardrail reminder
60/// 3. Skills — full skill instructions and tool metadata
61/// 4. Workspace — working directory
62/// 5. Bootstrap files — AGENTS, SOUL, TOOLS, IDENTITY, USER, BOOTSTRAP, MEMORY
63/// 6. Date & Time — timezone for cache stability
64/// 7. Runtime — host, OS, model
65///
66/// When `identity_config` is set to AIEOS format, the bootstrap files section
67/// is replaced with the AIEOS identity data loaded from file or inline JSON.
68///
69/// Daily memory files (`memory/*.md`) are NOT injected — they are accessed
70/// on-demand via `memory_recall` / `memory_search` tools.
71pub fn build_system_prompt(
72    workspace_dir: &std::path::Path,
73    model_name: &str,
74    tools: &[(&str, &str)],
75    skills: &[Skill],
76    identity_config: Option<&zeroclaw_config::schema::IdentityConfig>,
77    bootstrap_max_chars: Option<usize>,
78) -> String {
79    build_system_prompt_with_mode(
80        workspace_dir,
81        model_name,
82        tools,
83        skills,
84        identity_config,
85        bootstrap_max_chars,
86        false,
87        zeroclaw_config::schema::SkillsPromptInjectionMode::Full,
88        AutonomyLevel::default(),
89    )
90}

92pub fn build_system_prompt_with_mode(
93    workspace_dir: &std::path::Path,
94    model_name: &str,
95    tools: &[(&str, &str)],
96    skills: &[Skill],
97    identity_config: Option<&zeroclaw_config::schema::IdentityConfig>,
98    bootstrap_max_chars: Option<usize>,

100    skills_prompt_mode: zeroclaw_config::schema::SkillsPromptInjectionMode,

102) -> String {
103    let autonomy_cfg = zeroclaw_config::schema::AutonomyConfig {
104        level: autonomy_level,
105        ..Default::default()

107    build_system_prompt_with_mode_and_autonomy(
108        workspace_dir,
109        model_name,
110        tools,
111        skills,
112        identity_config,
113        bootstrap_max_chars,
114        Some(&autonomy_cfg),
115        native_tools,
116        skills_prompt_mode,
117        false,
118        0,
119        false,
120    )
121}

123#[allow(clippy::too_many_arguments)]
124pub fn build_system_prompt_with_mode_and_autonomy(
125    workspace_dir: &std::path::Path,
126    model_name: &str,
127    tools: &[(&str, &str)],
128    skills: &[Skill],
129    identity_config: Option<&zeroclaw_config::schema::IdentityConfig>,
130    bootstrap_max_chars: Option<usize>,
131    autonomy_config: Option<&zeroclaw_config::schema::AutonomyConfig>,
132    native_tools: bool,
133    skills_prompt_mode: zeroclaw_config::schema::SkillsPromptInjectionMode,
134    compact_context: bool,
135    max_system_prompt_chars: usize,
136    procedural_mode: bool,
137) -> String {
138    use std::fmt::Write;
139    let mut prompt = String::with_capacity(8192);
140
141    // ── 0. Execution-mode opener (top priority) ─────────────────
142    //
143    // Conversational mode (the framework default) injects the
144    // `## CRITICAL: No Tool Narration` block: tool calls are invisible
145    // infrastructure, the user only sees the final answer. Right for
146    // chat agents.
147    //
148    // Procedural mode skips that block and instead injects a
149    // `## Execution Mode: Procedural` block requiring the agent to
150    // emit each tool_call BEFORE producing analysis text and to never
151    // fabricate step output. Right for autonomous agents executing
152    // multi-step procedures (monitoring heartbeats, scheduled
153    // checklists). Added 2026-05-29 — see the procedural-mode bullet
154    // on `AgentConfig::system_prompt_mode` for the empirical
155    // motivation (Vigil's Phase 8 corpus rewrite proved the framework
156    // anti-narration directive was too loud for prompt-level
157    // workarounds to overcome).
158    if procedural_mode {
159        prompt.push_str(
160            "## Execution Mode: Procedural\n\n\
161             You are an autonomous agent executing a multi-step procedure. \
162             For each step defined in your workspace files (HEARTBEAT.md, AGENTS.md, the user's directive):\n\
163             1. Emit a `tool_call` FIRST — before any analysis text for that step.\n\
164             2. Wait for the `tool_result` to be returned to you as a tool message.\n\
165             3. Analyze the result, then proceed to the next step's `tool_call`.\n\n\
166             The step-by-step execution IS the deliverable. Tool emissions are NOT invisible infrastructure in procedural mode — they ARE the work.\n\n\
167             You may only report what an actual `tool_result` returned. \
168             If you did not emit a `tool_call` for a step, you do NOT have that step's data. \
169             Fabricating plausible numbers in place of an unmade tool call is forbidden and will be detected.\n\n",
170        );
171    } else {
172        prompt.push_str(
173            "## CRITICAL: No Tool Narration\n\n\
174             NEVER narrate, announce, describe, or explain your tool usage to the user. \
175             Do NOT say things like 'Let me check...', 'I will use http_request to...', \
176             'I'll fetch that for you', 'Searching now...', or 'Using the web_search tool'. \
177             The user must ONLY see the final answer. Tool calls are invisible infrastructure — \
178             never reference them. If you catch yourself starting a sentence about what tool \
179             you are about to use or just used, DELETE it and give the answer directly.\n\n",
180        );
181    }
182
183    // ── 0b. Tool Honesty (BOTH modes; no-fabrication is universal) ──
184    prompt.push_str(
185        "## CRITICAL: Tool Honesty\n\n\
186         - NEVER fabricate, invent, or guess tool results. If a tool returns empty results, say \"No results found.\"\n\
187         - If a tool call fails, report the error — never make up data to fill the gap.\n\
188         - When unsure whether a tool call succeeded, ask the user rather than guessing.\n\n",
189    );
190
191    // ── 1. Tooling ──────────────────────────────────────────────
192    if !tools.is_empty() {
193        prompt.push_str("## Tools\n\n");
194        if compact_context {
195            // Compact mode: tool names only, no descriptions/schemas
196            prompt.push_str("Available tools: ");
197            let names: Vec<&str> = tools.iter().map(|(name, _)| *name).collect();
198            prompt.push_str(&names.join(", "));
199            prompt.push_str("\n\n");
200        } else {
201            prompt.push_str("You have access to the following tools:\n\n");
202            for (name, desc) in tools {
203                let _ = writeln!(prompt, "- **{name}**: {desc}");
204            }
205            prompt.push('\n');
206        }
207    }
208
209    // ── 1b. Hardware (when gpio/arduino tools present) ───────────
210    let has_hardware = tools.iter().any(|(name, _)| {
211        *name == "gpio_read"
212            || *name == "gpio_write"
213            || *name == "arduino_upload"
214            || *name == "hardware_memory_map"
215            || *name == "hardware_board_info"
216            || *name == "hardware_memory_read"
217            || *name == "hardware_capabilities"
218    });
219    if has_hardware {
220        prompt.push_str(
221            "## Hardware Access\n\n\
222             You HAVE direct access to connected hardware (Arduino, Nucleo, etc.). The user owns this system and has configured it.\n\
223             All hardware tools (gpio_read, gpio_write, hardware_memory_read, hardware_board_info, hardware_memory_map) are AUTHORIZED and NOT blocked by security.\n\
224             When they ask to read memory, registers, or board info, USE hardware_memory_read or hardware_board_info — do NOT refuse or invent security excuses.\n\
225             When they ask to control LEDs, run patterns, or interact with the Arduino, USE the tools — do NOT refuse or say you cannot access physical devices.\n\
226             Use gpio_write for simple on/off; use arduino_upload when they want patterns (heart, blink) or custom behavior.\n\n",
227        );
228    }
229
230    // ── 1c. Action instruction (avoid meta-summary) ───────────────
231    //
232    // Procedural mode swaps the "give the FINAL ANSWER only" framing
233    // (which directly conflicts with HEARTBEAT.md / AGENTS.md style
234    // step-by-step procedures) for a directive that explicitly
235    // requires step-by-step tool emission. See the `## Execution Mode`
236    // block above for the motivation.
237    if procedural_mode {
238        prompt.push_str(
239            "## Your Task\n\n\
240             You are running an autonomous procedure (cron job, scheduled task, multi-step checklist). \
241             Follow the procedure step-by-step:\n\
242             - Emit `tool_call`s explicitly per the procedure.\n\
243             - Wait for each `tool_result` before proceeding to the next step.\n\
244             - The step-by-step structure IS the deliverable — do not collapse it into a single final answer.\n\n\
245             If the same agent receives an interactive query (Discord direct message, ad-hoc operator question), respond naturally and use tools as needed — but the procedural cron path is always step-by-step.\n\n",
246        );
247    } else if native_tools {
248        prompt.push_str(
249            "## Your Task\n\n\
250             When the user sends a message, respond naturally. Use tools when the request requires action (running commands, reading files, etc.).\n\
251             For questions, explanations, or follow-ups about prior messages, answer directly from conversation context — do NOT ask the user to repeat themselves.\n\
252             Do NOT: summarize this configuration, describe your capabilities, or output step-by-step meta-commentary.\n\n",
253        );
254    } else {
255        prompt.push_str(
256            "## Your Task\n\n\
257             When the user sends a message, ACT on it. Use the tools to fulfill their request.\n\
258             Do NOT: summarize this configuration, describe your capabilities, respond with meta-commentary, or output step-by-step instructions (e.g. \"1. First... 2. Next...\").\n\
259             Instead: emit actual <tool_call> tags when you need to act. Just do what they ask.\n\n",
260        );
261    }
262
263    // ── 2. Safety ───────────────────────────────────────────────
264    prompt.push_str("## Safety\n\n");
265    prompt.push_str("- Do not exfiltrate private data.\n");
266    if autonomy_config.map(|cfg| cfg.level) != Some(crate::security::AutonomyLevel::Full) {
267        prompt.push_str(
268            "- Do not run destructive commands without asking.\n\
269             - Do not bypass oversight or approval mechanisms.\n",
270        );
271    }
272    prompt.push_str("- Prefer `trash` over `rm` (recoverable beats gone forever).\n");
273    prompt.push_str(match autonomy_config.map(|cfg| cfg.level) {
274        Some(crate::security::AutonomyLevel::Full) => {
275            "- Respect the runtime autonomy policy: if a tool or action is allowed, execute it directly instead of asking the user for extra approval.\n\
276             - If a tool or action is blocked by policy or unavailable, explain that concrete restriction instead of simulating an approval dialog.\n"
277        }
278        Some(crate::security::AutonomyLevel::ReadOnly) => {
279            "- Respect the runtime autonomy policy: this runtime is read-only for side effects unless a tool explicitly reports otherwise.\n\
280             - If a requested action is blocked by policy, explain the restriction directly instead of simulating an approval dialog.\n"
281        }
282        _ => {
283            "- When in doubt, ask before acting externally.\n\
284             - Respect the runtime autonomy policy: ask for approval only when the current runtime policy actually requires it.\n\
285             - If a tool or action is blocked by policy or unavailable, explain that concrete restriction instead of simulating an approval dialog.\n"
286        }
287    });
288    prompt.push('\n');
289
290    // ── 3. Skills (full or compact, based on config) ─────────────
291    if !skills.is_empty() {
292        prompt.push_str(&crate::skills::skills_to_prompt_with_mode(
293            skills,
294            workspace_dir,
295            skills_prompt_mode,
296        ));
297        prompt.push_str("\n\n");
298    }
299
300    // ── 4. Workspace ────────────────────────────────────────────
301    let _ = writeln!(
302        prompt,
303        "## Workspace\n\nWorking directory: `{}`\n",
304        workspace_dir.display()
305    );
306
307    // ── 5. Bootstrap files (injected into context) ──────────────
308    prompt.push_str("## Project Context\n\n");
309
310    // Check if AIEOS identity is configured
311    if let Some(config) = identity_config {
312        if identity::is_aieos_configured(config) {
313            // Load AIEOS identity
314            match identity::load_aieos_identity(config, workspace_dir) {
315                Ok(Some(aieos_identity)) => {
316                    let aieos_prompt = identity::aieos_to_system_prompt(&aieos_identity);
317                    if !aieos_prompt.is_empty() {
318                        prompt.push_str(&aieos_prompt);
319                        prompt.push_str("\n\n");
320                    }
321                }
322                Ok(None) => {
323                    // No AIEOS identity loaded (shouldn't happen if is_aieos_configured returned true)
324                    // Fall back to OpenClaw bootstrap files
325                    let max_chars = bootstrap_max_chars.unwrap_or(BOOTSTRAP_MAX_CHARS);
326                    load_openclaw_bootstrap_files(&mut prompt, workspace_dir, max_chars);
327                }
328                Err(e) => {
329                    // Log error but don't fail - fall back to OpenClaw
330                    eprintln!(
331                        "Warning: Failed to load AIEOS identity: {e}. Using OpenClaw format."
332                    );
333                    let max_chars = bootstrap_max_chars.unwrap_or(BOOTSTRAP_MAX_CHARS);
334                    load_openclaw_bootstrap_files(&mut prompt, workspace_dir, max_chars);
335                }
336            }
337        } else {
338            // OpenClaw format
339            let max_chars = bootstrap_max_chars.unwrap_or(BOOTSTRAP_MAX_CHARS);
340            load_openclaw_bootstrap_files(&mut prompt, workspace_dir, max_chars);
341        }
342    } else {
343        // No identity config - use OpenClaw format
344        let max_chars = bootstrap_max_chars.unwrap_or(BOOTSTRAP_MAX_CHARS);
345        load_openclaw_bootstrap_files(&mut prompt, workspace_dir, max_chars);
346    }
347
348    // ── 6. Date & Time ──────────────────────────────────────────
349    let now = chrono::Local::now();
350    let _ = writeln!(
351        prompt,
352        "## Current Date & Time\n\n{} ({})\n",
353        now.format("%Y-%m-%d %H:%M:%S"),
354        now.format("%Z")
355    );
356
357    // ── 7. Runtime ──────────────────────────────────────────────
358    let host =
359        hostname::get().map_or_else(|_| "unknown".into(), |h| h.to_string_lossy().to_string());
360    let _ = writeln!(
361        prompt,
362        "## Runtime\n\nHost: {host} | OS: {} | Model: {model_name}\n",
363        std::env::consts::OS,
364    );
365
366    // ── 8. Channel Capabilities (skipped in compact_context mode) ──
367    if !compact_context {
368        prompt.push_str("## Channel Capabilities\n\n");
369        prompt.push_str("- You are running as a messaging bot. Your response is automatically sent back to the user's channel.\n");
370        prompt
371            .push_str("- You do NOT need to ask permission to respond — just respond directly.\n");
372        prompt.push_str(match autonomy_config.map(|cfg| cfg.level) {
373        Some(crate::security::AutonomyLevel::Full) => {
374            "- If the runtime policy already allows a tool, use it directly; do not ask the user for extra approval.\n\
375             - Never pretend you are waiting for a human approval click or confirmation when the runtime policy already permits the action.\n\
376             - If the runtime policy blocks an action, say that directly instead of simulating an approval flow.\n"
377        }
378        Some(crate::security::AutonomyLevel::ReadOnly) => {
379            "- This runtime may reject write-side effects; if that happens, explain the policy restriction directly instead of simulating an approval flow.\n"
380        }
381        _ => {
382            "- Ask for approval only when the runtime policy actually requires it.\n\
383             - If there is no approval path for this channel or the runtime blocks an action, explain that restriction directly instead of simulating an approval flow.\n"
384        }
385    });
386        prompt.push_str("- NEVER repeat, describe, or echo credentials, tokens, API keys, or secrets in your responses.\n");
387        prompt.push_str("- If a tool output contains credentials, they have already been redacted — do not mention them.\n");
388        prompt.push_str("- When a user sends a voice note, it is automatically transcribed to text. Your text reply is automatically converted to a voice note and sent back. Do NOT attempt to generate audio yourself — TTS is handled by the channel.\n");
389        prompt.push_str("- NEVER narrate or describe your tool usage. Do NOT say 'Let me fetch...', 'I will use...', 'Searching...', or similar. Give the FINAL ANSWER only — no intermediate steps, no tool mentions, no progress updates.\n\n");
390    } // end if !compact_context (Channel Capabilities)
391
392    // ── 9. Truncation (max_system_prompt_chars budget) ──────────
393    if max_system_prompt_chars > 0 && prompt.len() > max_system_prompt_chars {
394        // Truncate on a char boundary, keeping the top portion (identity + safety).
395        let mut end = max_system_prompt_chars;
396        // Ensure we don't split a multi-byte UTF-8 character.
397        while !prompt.is_char_boundary(end) && end > 0 {
398            end -= 1;
399        }
400        prompt.truncate(end);
401        prompt.push_str("\n\n[System prompt truncated to fit context budget]\n");
402    }
403
404    if prompt.is_empty() {
405        "You are ZeroClaw, a fast and efficient AI assistant built in Rust. Be helpful, concise, and direct."
406            .to_string()
407    } else {
408        prompt
409    }
410}

412/// Inject a single workspace file into the prompt with truncation and missing-file markers.
413fn inject_workspace_file(
414    prompt: &mut String,
415    workspace_dir: &std::path::Path,
416    filename: &str,
417    max_chars: usize,
418) {
419    use std::fmt::Write;
420
421    let path = workspace_dir.join(filename);
422    match std::fs::read_to_string(&path) {
423        Ok(content) => {
424            let trimmed = content.trim();
425            if trimmed.is_empty() {
426                return;
427            }
428            let _ = writeln!(prompt, "### {filename}\n");
429            // Use character-boundary-safe truncation for UTF-8
430            let truncated = if trimmed.chars().count() > max_chars {
431                trimmed
432                    .char_indices()
433                    .nth(max_chars)
434                    .map(|(idx, _)| &trimmed[..idx])
435                    .unwrap_or(trimmed)
436            } else {
437                trimmed
438            };
439            if truncated.len() < trimmed.len() {
440                prompt.push_str(truncated);
441                let _ = writeln!(
442                    prompt,
443                    "\n\n[... truncated at {max_chars} chars — use `read` for full file]\n"
444                );
445            } else {
446                prompt.push_str(trimmed);
447                prompt.push_str("\n\n");
448            }
449        }
450        Err(_) => {
451            // Missing-file marker (matches OpenClaw behavior)
452            let _ = writeln!(prompt, "### {filename}\n\n[File not found: {filename}]\n");
453        }
454    }
455}

487    /// the framework `## CRITICAL: No Tool Narration` opener. This
488    /// pins the conversational-mode contract so future edits cannot

505            "Tool Honesty block must be present in conversational mode"

575    /// Pin the bootstrap-file list shape so the system-prompt path does not
576    /// silently diverge from `crate::agent::personality::PERSONALITY_FILES`.
577    ///
578    /// Downstream agents (Vigil, in particular) write scheduled monitoring
579    /// procedures to HEARTBEAT.md and reference the file by name in their
580    /// cron prompts ("Run the heartbeat checklist defined in HEARTBEAT.md").
581    /// If HEARTBEAT.md is excluded from the bootstrap list, the model
582    /// receives the cron prompt with no HEARTBEAT.md content in context
583    /// and either fabricates a generic checklist or says it cannot find
584    /// the file. Vigil hit this on 2026-05-24 after the Phase 3 model swap
585    /// to Mistral made the silent-truncation failure mode visible.
586    #[test]
587    fn bootstrap_files_include_heartbeat_md() {
588        let dir = std::env::temp_dir().join(format!(
589            "system_prompt_bootstrap_test_{}",
590            uuid::Uuid::new_v4()
591        ));
592        std::fs::create_dir_all(&dir).unwrap();
593        std::fs::write(
594            dir.join("HEARTBEAT.md"),
595            "# Heartbeat\n\nSTEP 1 of 6: do the thing.\n",
596        )
597        .unwrap();
598
599        let mut prompt = String::new();
600        load_openclaw_bootstrap_files(&mut prompt, &dir, BOOTSTRAP_MAX_CHARS);
601
602        assert!(
603            prompt.contains("### HEARTBEAT.md"),
604            "load_openclaw_bootstrap_files must inject `### HEARTBEAT.md`; \
605             absent here means the bootstrap list was edited to drop \
606             HEARTBEAT.md and downstream cron agents will silently lose \
607             their procedural checklist. Full prompt was:\n{prompt}",
608        );
609        assert!(
610            prompt.contains("STEP 1 of 6: do the thing."),
611            "load_openclaw_bootstrap_files must inject HEARTBEAT.md body, \
612             not just the heading. Full prompt was:\n{prompt}",
613        );
614
615        let _ = std::fs::remove_dir_all(dir);
616    }
